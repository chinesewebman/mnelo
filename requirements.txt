# mnelo runtime requirements
#
# Minimum dependencies to run the memory system.
# Versions are pinned to the ones we test against; bump as needed.

# Vector library (8/6 plan §1 — 向量库必选二选一, usearch 是必选兜底后端, f16 精度)
usearch>=2.26

# sqlite-vec — 仅 legacy: vec0 虚拟表建表需要 (migrate/repair/init_db 工具还用);
# 运行时不再作为检索后端写 vectors 表。

# Embedder (BAAI/bge-small-zh-v1.5, CN-native)

# MCP server (Model Context Protocol over SSE)
# [8/29 fix] pin 升 2.x — PR #13 (5b5f2f7) 改了 dispatcher 用 mcp 2.x API
# (Server(name, on_list_tools=..., on_call_tool=...) constructor kwargs, Resource.uri=str,
# mime_type, CallToolResult.is_error, streamable_http_client) 但忘了 bump 这个 pin.
# CI 装 1.x → Server.__init__() got unexpected kwarg 'on_list_tools' → 18 ERROR + 20 FAILED.
mcp[cli]>=2.1.1,<3.0.0

# ASGI server for the SSE transport (transitive dep of mcp; pinned here for explicitness)

# ASGI framework (transitive dep of mcp; pinned for explicitness)

# TOML config loader — only needed on Python <3.11.
